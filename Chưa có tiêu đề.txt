local function setupM2DynamicListener(p)
    if p == LocalPlayer then return end
    local function onChar(char)
        if not char then return end
        local hum = char:WaitForChild("Humanoid", 5)
        local root = char:WaitForChild("HumanoidRootPart", 5)
        if not hum or not root then return end
        hum.AnimationPlayed:Connect(function(track)
            if not isBlockActive then return end
            local animId = track.Animation and track.Animation.AnimationId
            if animId == M2_TargetID then
                local myChar = LocalPlayer.Character
                if myChar and myChar:FindFirstChild("HumanoidRootPart") then
                    local myPos = myChar.HumanoidRootPart.Position
                    local cPos = root.Position
                    local dx, dy, dz = cPos.X - myPos.X, cPos.Y - myPos.Y, cPos.Z - myPos.Z
                    if (dx*dx + dy*dy + dz*dz) <= COMBAT_RANGE_SQ then
                        activeM2Count = activeM2Count + 1
                        rawUnblock()
                        local cleaned = false
                        local stoppedConn, endedConn, diedConn, timeoutTask
                        local function cleanup()
                            if cleaned then return end
                            cleaned = true
                            activeM2Count = math.max(0, activeM2Count - 1)
                            if stoppedConn then stoppedConn:Disconnect() end
                            if endedConn then endedConn:Disconnect() end
                            if diedConn then diedConn:Disconnect() end
                            if timeoutTask then task.cancel(timeoutTask) end
                        end
                        stoppedConn = track.Stopped:Connect(cleanup)
                        endedConn = track.Ended:Connect(cleanup)
                        diedConn = hum.Died:Connect(cleanup)
                        local animDuration = (track.Length > 0 and track.Length) or 1.5
                        timeoutTask = task.delay(animDuration + 0.1, cleanup)
                    end
                end
            end
        end)
    end
    if p.Character then onChar(p.Character) end
    p.CharacterAdded:Connect(onChar)
end

local function Valid(p)
    if not p or p == LocalPlayer or Allies[p.Name] then return false end
    local c = p.Character
    if not c then return false end
    local r = c:FindFirstChild("HumanoidRootPart")
    local h = c:FindFirstChildOfClass("Humanoid")
    return r and h and h.Health > 0
end

local function FindNearest(range)
    local mr = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not mr then return nil end
    local best, bd = nil, range or 100
    for _, p in ipairs(Players:GetPlayers()) do
        if Valid(p) then
            local r = p.Character:FindFirstChild("HumanoidRootPart")
            if r then
                local d = (mr.Position - r.Position).Magnitude
                if d < bd then bd = d; best = p end
            end
        end
    end
    return best
end

local function FindByName(name)
    if not name or name == "" then return nil end
    local low = string.lower(name)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and string.find(string.lower(p.Name), low, 1, true) then return p end
    end
    return nil
end

local function GetTool()
    local c = LocalPlayer.Character
    if not c then return nil end
    local t = c:FindFirstChildOfClass("Tool")
    if t then return t end
    local bp = LocalPlayer:FindFirstChild("Backpack")
    if bp then
        for _, x in ipairs(bp:GetChildren()) do
            if x:IsA("Tool") then x.Parent = c; return x end
        end
    end
    return nil
end

local function EzyFastAttack(target)
    local c = LocalPlayer.Character
    if not c then return end
    local hum = c:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return end
    for _, track in ipairs(hum:GetPlayingAnimationTracks()) do
        pcall(function()
            if track.Priority == Enum.AnimationPriority.Action or track.Priority == Enum.AnimationPriority.Action2 then
                track:Stop(0.1)
            end
        end)
    end
    local tool = c:FindFirstChildOfClass("Tool")
    if tool then pcall(function() tool:Activate() end) end
    if clientSideHitbox and target then
        pcall(function() clientSideHitbox:FireServer(target, buffer.fromstring("10069.74")) end)
    end
    if combatRemote then
        pcall(function() combatRemote:FireServer("M1") end)
        pcall(function() combatRemote:FireServer("Attack") end)
    end
    for _, r in ipairs(Remotes) do
        pcall(function() r:FireServer("M1") end)
        pcall(function() r:FireServer() end)
    end
    for _, r in ipairs(ManualRemotes) do
        pcall(function() r:FireServer("M1") end)

        pcall(function() r:FireServer() end)
    end

local function IsMonster(model)
    if not model or not model:IsA("Model") then return false end
    local hum = model:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    if model == LocalPlayer.Character then return false end
    if Players:GetPlayerFromCharacter(model) then return false end
    return (model:FindFirstChild("HumanoidRootPart") or model.PrimaryPart) ~= nil
end

local function GetMonsterRoot(model)
    return model:FindFirstChild("HumanoidRootPart") or model.PrimaryPart
end

local function ScanMonsters()
    local list = {}
    for _, v in ipairs(workspace:GetDescendants()) do
        if v:IsA("Model") and IsMonster(v) then table.insert(list, v) end
    end
    return list
end

local function FindTargetByNamePartial(partial)
    if not partial or partial == "" then return nil end
    local low = string.lower(partial)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and string.find(string.lower(p.Name), low, 1, true) then return p end
    end
    return nil
end

local function ClaimOwnership(model)
    pcall(function()
        for _, part in ipairs(model:GetDescendants()) do
            if part:IsA("BasePart") and part.SetNetworkOwner then
                part:SetNetworkOwner(LocalPlayer)
            end
        end
    end)
end

local function CollectAllMonsters()
    local myChar = LocalPlayer.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end
    local found = ScanMonsters()
    local count = 0
    for _, model in ipairs(found) do
        if count >= Config.MonsterMax then break end
        local root = GetMonsterRoot(model)
        if root then
            ClaimOwnership(model)
            local angle = (count / math.max(1, #found)) * math.pi * 2
            local pos = myRoot.Position + Vector3.new(
                math.cos(angle) * Config.MonsterOrbitRadius,
                Config.MonsterOrbitHeight,
                math.sin(angle) * Config.MonsterOrbitRadius
            )
            pcall(function()
                model:PivotTo(CFrame.new(pos))
                local hum = model:FindFirstChildOfClass("Humanoid")
                if hum then hum.WalkSpeed = 60 end
            end)
            monsters[model] = true
            count = count + 1
        end
    end
end

local function SummonAgainstTarget()
    local target = Config.MonsterTarget
    if not target or not target.Character then return end
    local tRoot = target.Character:FindFirstChild("HumanoidRootPart")
    if not tRoot then return end
    local found = ScanMonsters()
    local count = 0
    for _, model in ipairs(found) do
        if count >= Config.MonsterMax then break end
        local root = GetMonsterRoot(model)
        local hum = model:FindFirstChildOfClass("Humanoid")
        if root and hum then
            ClaimOwnership(model)
            if Config.MonsterMode == "TP" then
                local off = Vector3.new(math.random(-3, 3), 2, math.random(-3, 3))
                pcall(function() model:PivotTo(CFrame.new(tRoot.Position + off)) end)
            end
            monsters[model] = true
            count = count + 1
        end
    end
end

                if dist > 2000 then return end
                local tp = tRoot.Position

                if Config.ORMode == "DASH" then
                    local now = tick()
                    local function groundAt(x, z, baseY)
                        local rp = RaycastParams.new()
                        rp.FilterType = Enum.RaycastFilterType.Exclude
                        rp.FilterDescendantsInstances = {myC, tc}
                        local hit = workspace:Raycast(Vector3.new(x, baseY + 20, z), Vector3.new(0, -80, 0), rp)
                        return hit and hit.Position.Y or (baseY - 3)
                    end
                    local function tpToTarget()
                        local behind = tp - tRoot.CFrame.LookVector * Config.DashOrbitRadius
                        pcall(function()
                            myRoot.CFrame = CFrame.lookAt(Vector3.new(behind.X, tp.Y, behind.Z), tp)
                            myRoot.AssemblyLinearVelocity = Vector3.zero
                        end)
                    end
                    if dashPhase == "idle" then
                        dashSavedPos = myRoot.CFrame
                        dashOrbitAngle = math.random() * math.pi * 2
                        tpToTarget()
                        dashPhase = "attack"
                        dashTimer = now + Config.DashAttackDuration
                        dashNextHit = 0
                    elseif dashPhase == "attack" then
                        dashOrbitAngle = dashOrbitAngle + math.rad(Config.DashOrbitSpeed) * 0.02
                        local px = tp.X + math.cos(dashOrbitAngle) * Config.DashOrbitRadius
                        local pz = tp.Z + math.sin(dashOrbitAngle) * Config.DashOrbitRadius
                        local gy = groundAt(px, pz, tp.Y)
                        local desired = Vector3.new(px, gy + Config.DashGroundOffset, pz)
                        pcall(function()
                            myRoot.CFrame = CFrame.lookAt(desired, Vector3.new(tp.X, desired.Y, tp.Z))
                            myRoot.AssemblyLinearVelocity = Vector3.zero
                        end)
                        if now >= dashNextHit then
                            dashNextHit = now + Config.DashHitRate
                            HitAll(Config.Target)
                        end
                        if now >= dashTimer then
                            if dashSavedPos then
                                pcall(function()
                                    myRoot.CFrame = dashSavedPos
                                    myRoot.AssemblyLinearVelocity = Vector3.zero
                                end)
                            end
                            dashPhase = "wait"
                            dashTimer = now + Config.DashWaitDuration
                        end
                    elseif dashPhase == "wait" then
                        if now >= dashTimer then
                            dashOrbitAngle = math.random() * math.pi * 2
                            tpToTarget()
                            dashPhase = "attack"
                            dashTimer = now + Config.DashAttackDuration
                            dashNextHit = 0
                        end
                    end
                elseif Config.ORMode == "EZY" then
                    local behind = tp - tRoot.CFrame.LookVector * Config.DashOrbitRadius
                    local dir = behind - myRoot.Position
                    if dir.Magnitude > 1.5 then
                        myRoot.AssemblyLinearVelocity = dir.Unit * (Config.ORWalkSpeed * 10)
                    else
                        myRoot.AssemblyLinearVelocity = Vector3.zero
                        pcall(function() myRoot.CFrame = CFrame.lookAt(myRoot.Position, tp) end)
                    end
                    EzyFastAttack(Config.Target)
                else
                    local fp
                    local dcf = tRoot.CFrame
                    if Config.ORMode == "FOLLOW" then fp = tp - dcf.LookVector * Config.ORDistance
                    elseif Config.ORMode == "BEHIND" then fp = tp - dcf.LookVector * Config.ORDistance
                    elseif Config.ORMode == "ORBIT" then
                        Config.orbitAngle = Config.orbitAngle + math.rad(Config.ORRotSpeed) * 0.05
                        fp = tp + Vector3.new(math.cos(Config.orbitAngle)*Config.ORDistance, Config.ORHeight, math.sin(Config.orbitAngle)*Config.ORDistance)

                    pcall(function()
                        myRoot.CFrame = CFrame.lookAt(fp + Vector3.new(0, Config.ORHeight, 0), tp + Vector3.new(0, Config.ORHeight, 0))
                        myRoot.AssemblyLinearVelocity = Vector3.zero
                    end)
                    if dist <= Config.ORNear then HitAll(Config.Target) end
                end
            end)
        end
    end
end)

-- Anti chan
local isAntiChanRunning = false
local antiChanLastHealth = 100
local function findNearestEnemy()
    local ch = LocalPlayer.Character
    local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local nearest, nearestDist = nil, math.huge
    for _, pl in pairs(Players:GetPlayers()) do
        if pl ~= LocalPlayer and not table.find(ProtectedPlayers, pl) and not Allies[pl.Name] then
            local tch = pl.Character
            local trp = tch and tch:FindFirstChild("HumanoidRootPart")
            local th = tch and tch:FindFirstChildOfClass("Humanoid")
            if trp and th and th.Health > 0 then
                local d = (hrp.Position - trp.Position).Magnitude
                if d < nearestDist then nearest = pl; nearestDist = d end
            end
        end
    end
    return nearest
end

local function doAntiChanCounter(attacker)
    if isAntiChanRunning then return end
    local ch = LocalPlayer.Character
    local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
    local hum = ch and ch:FindFirstChildOfClass("Humanoid")
    if not (hrp and hum and hum.Health > 0) then return end
    isAntiChanRunning = true
    local bv = Instance.new("BodyVelocity", hrp)
    bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    bv.Velocity = Vector3.new(math.random(-150,150), math.random(100,200), math.random(-150,150))
    for _, part in pairs(ch:GetDescendants()) do
        if part:IsA("BasePart") then part.CanCollide = false end
    end
    task.wait(0.45)
    bv:Destroy()
    local target = attacker or findNearestEnemy()
    local trp = target and target.Character and target.Character:FindFirstChild("HumanoidRootPart")
    if trp and hrp then
        hrp.CFrame = trp.CFrame * CFrame.new(0, 0, 3)
        hrp.AssemblyLinearVelocity = Vector3.zero
    end
    for _, part in pairs(ch:GetDescendants()) do
        if part:IsA("BasePart") then part.CanCollide = true end
    end
    if combatRemote then
        for _ = 1, 3 do
            pcall(function()
                combatRemote:FireServer("M1")
                combatRemote:FireServer("M2")
                combatRemote:FireServer("Kick")
                combatRemote:FireServer("Block")
            end)
            task.wait(0.08)
        end
    end
    isAntiChanRunning = false
end

local function setupAntiChanListener(char)
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    antiChanLastHealth = hum.Health
    hum.HealthChanged:Connect(function(newH)
        if State.SAnhAntiChan and not isAntiChanRunning and newH < antiChanLastHealth and newH > 0 then
            task.spawn(function() doAntiChanCounter(findNearestEnemy()) end)
        end
        antiChanLastHealth = newH
    end)
end
LocalPlayer.CharacterAdded:Connect(function(c) task.wait(0.5); setupAntiChanListener(c) end)
if LocalPlayer.Character then setupAntiChanListener(LocalPlayer.Character) end

                            if (item.part.Position - root.Position).Magnitude <= Config.FarmRadius then
                                table.insert(objs, item)
                            end
                        end
                    end)
                end
                for _, item in ipairs(objs) do
                    if not State.Farm then break end
                    for i = 1, Config.FarmClicks do
                        if not State.Farm then break end
                        if item.type == "prompt" then
                            pcall(function()
                                item.obj:InputHoldBegin()
                                task.wait(0.08)
                                item.obj:InputHoldEnd()
                            end)
                        elseif item.type == "click" then
                            pcall(function() fireclickdetector(item.obj) end)
                        end
                        task.wait(0.15)
                    end
                end
                if State.Farm then
                    task.wait(Config.FarmDelay)
                    local oldChar = LocalPlayer.Character
                    pcall(function()
                        if oldChar then
                            local h = oldChar:FindFirstChildOfClass("Humanoid")
                            if h then h.Health = 0 end
                        end
                    end)
                    local wT = 0
                    while State.Farm and wT < 10 do
                        task.wait(0.2)
                        wT = wT + 0.2
                        local nc = LocalPlayer.Character
                        if nc and nc ~= oldChar then
                            local nh = nc:FindFirstChildOfClass("Humanoid")
                            if nh and nh.Health > 0 then break end
                        end
                    end
                    if State.Farm and Config.FarmPos then
                        task.wait(0.5)
                        for i = 1, 30 do
                            if not State.Farm then break end
                            local r2 = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                            if r2 then
                                pcall(function() r2.CFrame = Config.FarmPos; r2.AssemblyLinearVelocity = Vector3.zero end)
                            end
                            task.wait(0.1)
                        end
                    end
                end
                Config.FarmRunning = false
            end
        end
    end
end)

local saOrbitAngle = 0
local saFlyAngle = 0
local saHiddenPlatform = nil

RunService.RenderStepped:Connect(function(dt)

                    if pl ~= LocalPlayer and pl.Character and string.find(string.lower(pl.Name), string.lower(name), 1, true) then

                    end
                end

                camera.CFrame = CFrame.new(camera.CFrame.Position, target.Character.Head.Position)
            end
        end

        -- HIDDEN MODE
        if State.SAnhHiddenMode then
            local ch = LocalPlayer.Character
            local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
            if hrp then
                if not saHiddenPlatform or not saHiddenPlatform.Parent then
                    local rp = RaycastParams.new()
                    rp.FilterType = Enum.RaycastFilterType.Exclude
                    rp.FilterDescendantsInstances = {ch}
                    local res = workspace:Raycast(Vector3.new(hrp.Position.X, 10000, hrp.Position.Z), Vector3.new(0, -20000, 0), rp)
                    local groundY = res and res.Position.Y or 0
                    saHiddenPlatform = Instance.new("Part")
                    saHiddenPlatform.Size = Vector3.new(30, 1, 30)
                    saHiddenPlatform.Position = Vector3.new(hrp.Position.X, groundY + 250, hrp.Position.Z)
                    saHiddenPlatform.Anchored = true
                    saHiddenPlatform.Transparency = 0.5
                    saHiddenPlatform.Material = Enum.Material.Neon
                    saHiddenPlatform.Color = Color3.fromRGB(0, 200, 255)
                    saHiddenPlatform.CanCollide = true
                    saHiddenPlatform.Parent = workspace
                end
                hrp.CFrame = CFrame.new(Vector3.new(hrp.Position.X, saHiddenPlatform.Position.Y + 3.5, hrp.Position.Z)) * hrp.CFrame.Rotation
                hrp.AssemblyLinearVelocity = Vector3.zero
            end
        else
            if saHiddenPlatform then saHiddenPlatform:Destroy(); saHiddenPlatform = nil end
        end

        -- HITBOX ORBIT (S_anh)
        if State.SAnhHitboxOrbit then
            local ch = LocalPlayer.Character
            local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
            if hrp then
                local target
                local name = MonsterTargetBox.Text
                for _, pl in ipairs(Players:GetPlayers()) do
                    if pl ~= LocalPlayer and pl.Character and string.find(string.lower(pl.Name), string.lower(name), 1, true) then
                        target = pl break
                    end
                end
                if target and target.Character then
                    local tHead = target.Character:FindFirstChild("Head")
                    local tHum = target.Character:FindFirstChildOfClass("Humanoid")
                    if tHead and tHum and tHum.Health > 0 then
                        local center = tHead.Position + Vector3.new(0, Config.SAHitboxHeight, 0)
                        local edgeRadius = math.max(Config.SAHitboxSize * 0.5, 1) + 0.5 + Config.SAOrbitRadius
                        saOrbitAngle = (saOrbitAngle + Config.SAOrbitSpeed * dt) % (math.pi * 2)
                        local ox = math.cos(saOrbitAngle) * edgeRadius
                        local oz = math.sin(saOrbitAngle) * edgeRadius
                        local orbitPos = center + Vector3.new(ox, 0, oz)
                        local lookDir = (center - orbitPos)
                        if lookDir.Magnitude < 0.01 then lookDir = Vector3.new(0, 0, 1) end
                        lookDir = lookDir.Unit
                        hrp.CFrame = CFrame.lookAt(orbitPos, orbitPos + lookDir)
                        hrp.AssemblyLinearVelocity = Vector3.zero
                        hrp.AssemblyAngularVelocity = Vector3.zero
                    end
                end
            end
        end

        -- AUTO FLY
        if State.SAnhAutoFly then
            local ch = LocalPlayer.Character
            local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
            if hrp then
                saFlyAngle = saFlyAngle + (Config.SAFlySpeed / 300) * dt
                local x = math.cos(saFlyAngle) * 300
                local z = math.sin(saFlyAngle) * 300
                hrp.CFrame = CFrame.new(Vector3.new(x, 50, z), Vector3.new(x + (-x), 50, z + (-z)))
            end
        end

        -- S_anh spin
        if State.SAnhSpin then
            local c = LocalPlayer.Character
            local hrp = c and c:FindFirstChild("HumanoidRootPart")
            if hrp then hrp.CFrame = hrp.CFrame * CFrame.Angles(0, math.rad(Config.SASpinSpeed * dt), 0) end
        end
    end)
end)

                                        end
                                    end
                                end
                                if m1Detected then break end
                            end
                        end
                        if m1Detected then
                            lastM1Time = tick()
                            rawBlock()
                        else
                            if isBlocking and (tick() - lastM1Time) > BUFFER_TIME then rawUnblock() end
                        end
                    end
                end
            end
        end
        if Config.AutoReset then
            local myChar = LocalPlayer.Character
            local myHum = myChar and myChar:FindFirstChildOfClass("Humanoid")
            if myHum and myHum.Health > 0 then
                if State.ORLock or State.M1Spam or State.Spam then

                else lastResetTimer = tick() end
